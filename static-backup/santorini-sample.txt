  if (!window.gsap) return;
  if (matchMedia("(prefers-reduced-motion: reduce)").matches) return;
  if (matchMedia("(pointer: coarse)").matches) return; // mobile e native scroll

  const html = document.documentElement;
  html.style.scrollBehavior = "auto";

  const SPEED = 1;       // wheel e koto dur jabe
  const DURATION = 1.1;  // beshi = aro smooth
  const pos = { y: window.scrollY };
  let target = window.scrollY;
  let animating = false;

  const maxScroll = () => html.scrollHeight - window.innerHeight;
  const stop = () => { gsap.killTweensOf(pos); animating = false; };

  window.addEventListener("wheel", (e) => {
    if (e.ctrlKey || e.defaultPrevented) return;
    if (e.target.closest && e.target.closest("[data-native-scroll]")) return;
    e.preventDefault();

    let d = e.deltaY;
    if (e.deltaMode === 1) d *= 16;
    else if (e.deltaMode === 2) d *= window.innerHeight;

    if (!animating) pos.y = target = window.scrollY;
    target = gsap.utils.clamp(0, maxScroll(), target + d * SPEED);
    animating = true;

    gsap.to(pos, {
      y: target,
      duration: DURATION,
      ease: "power3.out",
      overwrite: true,
      onUpdate: () => window.scrollTo(0, pos.y),
      onComplete: () => { animating = false; }
    });
  }, { passive: false });

  window.addEventListener("mousedown", stop);
  window.addEventListener("keydown", stop);
  document.addEventListener("click", (e) => {

  window.addEventListener("scroll", () => {
    if (!animating) pos.y = target = window.scrollY;
